cmake_minimum_required(VERSION 3.20)
project(pyctr_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Eigen3 REQUIRED NO_MODULE)
add_library(pyctr_core INTERFACE)
target_include_directories(pyctr_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(pyctr_core INTERFACE Eigen3::Eigen)

# One library per component, src/<name>.cpp with include/pyctr/<name>.hpp. Missing components are skipped.
set(PYCTR_COMPONENTS rod robot rod_dynamics robot_dynamics config)
foreach(c ${PYCTR_COMPONENTS})
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/${c}.cpp)
    add_library(pyctr_${c} STATIC src/${c}.cpp)
    target_link_libraries(pyctr_${c} PUBLIC pyctr_core)
  endif()
endforeach()
if(TARGET pyctr_config)
  find_package(yaml-cpp REQUIRED)
  target_link_libraries(pyctr_config PUBLIC yaml-cpp::yaml-cpp)
endif()

# Python package pyctr_cpp in the build directory: one extension module _<name> per component.
# cmake --install (used by pip install ./cpp, see pyproject.toml) copies the package to <prefix>/pyctr_cpp.
option(PYCTR_PYTHON "Build the Python package pyctr_cpp" ON)
if(PYCTR_PYTHON)
  find_package(Python 3 COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  foreach(c ${PYCTR_COMPONENTS})
    if(TARGET pyctr_${c} AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/python/bind_${c}.cpp)
      pybind11_add_module(_${c} python/bind_${c}.cpp)
      target_link_libraries(_${c} PRIVATE pyctr_${c})
      target_include_directories(_${c} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/python)
      set_target_properties(_${c} PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/pyctr_cpp)
      install(TARGETS _${c} LIBRARY DESTINATION pyctr_cpp)
    endif()
  endforeach()
  configure_file(python/pyctr_cpp/__init__.py ${CMAKE_BINARY_DIR}/pyctr_cpp/__init__.py COPYONLY)
  install(FILES python/pyctr_cpp/__init__.py DESTINATION pyctr_cpp)
endif()

# C++ executables with assert-based checks: tests/<name>.cpp, run with ctest
option(PYCTR_TESTS "Build the C++ test executables" ON)
enable_testing()
if(PYCTR_TESTS)
  file(GLOB test_sources ${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp)
endif()
foreach(test_source ${test_sources})
  get_filename_component(test_name ${test_source} NAME_WE)
  add_executable(${test_name} ${test_source})
  target_link_libraries(${test_name} PRIVATE pyctr_core)
  foreach(c ${PYCTR_COMPONENTS})
    if(TARGET pyctr_${c})
      target_link_libraries(${test_name} PRIVATE pyctr_${c})
    endif()
  endforeach()
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
